2.2 SEQUENCE DIAGRAM (UML)
--------------------------------------------------------------------

mermaid:

sequenceDiagram
    actor R as Recruiter
    participant FE as Frontend (React)
    participant BE as Backend API (Express)
    participant NLP as AI/Scoring Service
    participant DB as MongoDB

    R->>FE: Submit job details + resume files
    FE->>BE: POST /api/analyze/batch
    BE->>BE: Validate token and payload
    BE->>NLP: Parse and analyze resumes
    NLP-->>BE: Score + matched/missing skills + classification
    BE->>DB: Save analysis result
    DB-->>BE: Save confirmation
    BE-->>FE: Return structured analysis
    FE-->>R: Render rankings, insights, shortlist actions

      +exportShortlistReport()

2.11 ENTITY RELATIONSHIP DIAGRAM (ERD)

erDiagram

    ANALYSIS ||--o{ RESUME : contains
    ANALYSIS ||--o{ SHORTLIST_ITEM : produces

      string id PK
      string email
      string passwordHash
      string role
      datetime createdAt
    }

    ANALYSIS {
      string id PK
      string userId FK
      string jobTitle
      string jobDescription
      float score
      string classification
      datetime createdAt
    }

    RESUME {
      string id PK
      string analysisId FK
      string candidateName
      string fileName
      text extractedText
    }

    SHORTLIST_ITEM {
      string id PK
      string analysisId FK
